function(lab_executable name)
    add_executable(${name} ${ARGN})
    set_target_properties(${name} PROPERTIES OUTPUT_NAME "lib${name}.so" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    target_link_libraries(${name} PRIVATE henge_core)
endfunction()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(TEST classify operands)
    if("${TEST}" STREQUAL "classify")
        set(INPUT classify_cases.s)
    else()
        set(INPUT operand_cases.s)
    endif()
    set(ASM_OUTPUT "${CMAKE_BINARY_DIR}/${TEST}_cases.S")
    add_custom_command(OUTPUT "${ASM_OUTPUT}" COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/convert_asm.py" "${ROOT}/tests/${INPUT}" "${ASM_OUTPUT}" DEPENDS "${ROOT}/tests/${INPUT}")
    set_source_files_properties("${ASM_OUTPUT}" PROPERTIES COMPILE_OPTIONS "-march=armv8.4-a")
    lab_executable(test_${TEST} "${ROOT}/tests/test_${TEST}.c" "${ASM_OUTPUT}")
endforeach()
